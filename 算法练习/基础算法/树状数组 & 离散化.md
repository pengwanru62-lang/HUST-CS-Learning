##### P3374 [【模板】树状数组 1](https://www.luogu.com.cn/problem/P3374 "【模板】树状数组 1")（单点修改+区间查询）类似前缀和
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 5e5 + 5;
int tree[N];    //tree[i]表示从i-lowbit(i)+1到i的区间之和

int n,m;

int lowbit(int x){
    return x & -x;
}

void add(int x, int value){
    while(x<=n){
        tree[x] += value;
        x += lowbit(x);
    }
}

int query(int x){
    int sum = 0;
    while(x>0){
        sum+=tree[x];

        x-=lowbit(x);
    }
    return sum;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;

    for(int i=1; i<=n; i++){
        int a;
        cin >> a;

        add(i,a);
    }

    while(m--){
        int op,x;
        cin >> op >> x;

        if(op == 1){
            int k;
            cin >> k;

            add(x,k);
            
        }else if(op == 2){
            int y;
            cin >> y;
            cout << query(y) - query(x-1) << '\n';
        }
    }
    return 0;
}
```

##### P3368 [【模板】树状数组 2](https://www.luogu.com.cn/problem/P3368 "【模板】树状数组 2")
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 5e5+5;
int tree[N];
int n,m;

int lowbit(int x){
    return x & -x;
}

void add(int x, int value){
    while(x <= n){
        tree[x] += value;
        x += lowbit(x);
    }
}

int query(int x){
    int sum=0;
    while(x>0){
        sum+=tree[x];
        x-=lowbit(x);
    }
    return sum;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;

    int last = 0;
    for(int i=1; i<=n; i++){
        int a;
        cin >> a;
        add(i,a-last);
        last = a;    //存差分数组
    }

    while(m--){
        int op;
        cin >> op;
        if(op == 1){
            int x,y,k;
            cin >> x >> y >> k;
            add(x,k);
            if(y+1<=n){
                add(y+1, -k);
            }
            
        }else if(op == 2){
            int x;
            cin >> x;
            cout << query(x) << "\n";
        }
    }
    return 0;
}
```

#### P1908 [逆序对](https://www.luogu.com.cn/problem/P1908 "逆序对")
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 5e5+5;
int a[N];
int tree[N];
int n;
vector<int> b;

int lowbit(int x){
    return x & -x;
}

void add(int x, int value){
    while(x<=n){
        tree[x]+=value;
        x+=lowbit(x);
    }
}

int query(int x){
    int sum = 0;
    while(x>0){
        sum+=tree[x];
        x-=lowbit(x);
    }
    return sum;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    for(int i=1; i<=n; i++){
        cin >> a[i];    //储存a[i]

        //复制一份用于排序和离散化
        b.push_back(a[i]);       
    }

    sort(b.begin(), b.end());    //排序

    b.erase(unique(b.begin(),b.end()),b.end());    //去重

    long long ans = 0;

    for(int i=n; i>0; i--){
        int rank = lower_bound(b.begin(),b.end(),a[i])-b.begin()+1;    //lower_bound代替find函数，+1保证tree从1开始
        ans+=query(rank - 1);

        add(rank,1);
    }
    cout << ans;
    return 0;
}
```

#### P1637 [三元上升子序列](https://www.luogu.com.cn/problem/P1637 "三元上升子序列")
```cpp
#include<bits/stdc++.h>
using namespace std;

int n;

const int N = 3e4 + 5;
int a[N];
vector<int> b;
int l[N];
int r[N];

int tree[N]; //tree[i]表示rank为i的有几个

int lowbit(int x){
    return x & -x;
}

void add(int x, int value){
    while(x<=n){
        tree[x] += value;
        x += lowbit(x);
    }
}

//查询已经加入的数字中<=x的有多少个
int query(int x){
    int sum = 0;
    while(x>0){
        sum += tree[x];
        x -= lowbit(x);
    }
    return sum;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;

    for(int i=1; i<=n; i++){
        cin >> a[i];
        b.push_back(a[i]);    //复制一遍数组
    }

    //排序
    sort(b.begin(), b.end());

    //去重
    b.erase(unique(b.begin(), b.end()), b.end());

    //从左到右扫描统计
    for(int i=1; i<=n; i++){
        int rank = lower_bound(b.begin(), b.end(), a[i]) - b.begin()+1;

        l[i] = query(rank - 1);

        add(rank, 1);
    }

    //重置tree
    memset(tree, 0, sizeof(tree));

    //从右向左扫描大于当前的数
    for(int i=n; i>0; i--){
        int rank = lower_bound(b.begin(), b.end(), a[i]) - b.begin()+1;

        r[i] = (n-i) -query(rank);

        add(rank,1);
    }

    //统计
    long long ans = 0;
    for(int i=1; i<=n; i++){
        ans +=1LL * r[i]*l[i];
    }
    cout << ans;
    return 0;
}
```
