### 二分答案
#### P2440 [木材加工](https://www.luogu.com.cn/problem/P2440 "木材加工")
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e5 + 5;
int a[N];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;

    int l = 1;
    int r = 0;
	
	//输入数据的同时统计最大值从而能找到木端长的最大值
    for(int i = 1; i <= n; i++){
        cin >> a[i];
        r = max(r, a[i]);
    }

    int ans = 0;

    while(l <= r){
        int mid = l + (r - l) / 2;

        long long sum = 0;  //sum必须用long long从而防止溢出

        for(int i = 1; i <= n; i++){
            sum += a[i] / mid;

            
            if(sum >= k)  //已经够了那么就没有必要 再统计
                break;
        }

        if(sum >= k){  //注意更新逻辑，只要sum>=k就满足条件不一定要相等
            ans = mid;
            l = mid + 1;
        }else{
            r = mid - 1;
        }
    }

    cout << ans << '\n';

    return 0;
}
```

#### P2678 [[NOIP 2015 提高组] 跳石头](https://www.luogu.com.cn/problem/P2678 "[NOIP 2015 提高组] 跳石头")
```cpp
#include<bits/stdc++.h>
using namespace std;

//删除一块石头会影响两次距离，所以不可以使用差分
const int N = 50005;
int d[N];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int len, n, m;
    cin >> len >> n >> m;

    for(int i=1; i<=n; i++) cin >> d[i];

    d[n+1] = len;//注意加入最后一块石头才可以判断是否删除最后一块石头
    //使用二分法，最短距离的最大值，可可可|不可不可不可，我们的目标是找到变化的那个值
    //直接二分答案，那么初始化就从距离两端来确定
    int l = 1;//起点和中点不重合，那么最小值至少为1；
    int r = len;
    int ans = 0;

    while(l<=r){
        int mid = l + (r-l)/2;
        int removed = 0;
        int last = 0;

        //每一个位置都要检查来判断mid是不是满足要求
        for(int i=1; i<=n+1; i++){
            if(d[i]-last < mid){
                removed++;
            }else{
                last = d[i];
            }
        }
        if(removed <= m){
            ans = mid;
            l = mid + 1;
        }else{
            r = mid - 1;
        }
    }

    cout << ans << "\n";
    return 0;
}
```
