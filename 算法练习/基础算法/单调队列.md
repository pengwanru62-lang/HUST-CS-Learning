#### p1886 [【模板】单调队列 / 滑动窗口](https://www.luogu.com.cn/problem/P1886 "【模板】单调队列 / 滑动窗口")
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 1e6+5;
int a[N];    //窗口里面的具体值
deque<int> q;    //存放最小值队列下标
deque<int> p;    //存放最大值下标

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,k;
    cin >> n >> k;

    for(int i=1; i<=n; i++){
        cin >> a[i];

        //对于每一个刚刚准备进队的元素，首先保障队列的单调性
        while(!q.empty() && a[q.back()] >= a[i])
            q.pop_back();

        //新进队的元素进队
        q.push_back(i);

        //删除过期的元素
        while(!q.empty() && q.front() < i-k+1)
            q.pop_front();

        //输出最小值
        if(i>=k)
            cout << a[q.front()] << " ";
    }
    cout << "\n";

    for(int i=1; i<=n; i++){
        //对于每一个准备新入队的元素，首先需要保证队列的单调性
        while(!p.empty() && a[p.back()] <= a[i])
            p.pop_back();

        //新元素角标入队
        p.push_back(i);

        //删除过期元素
        while(!p.empty() && p.front()<i-k+1)
            p.pop_front();

        //输出最小值
        if(i>=k)
            cout << a[p.front()] << " ";
    }
    cout << "\n";
    return 0;
}
```
