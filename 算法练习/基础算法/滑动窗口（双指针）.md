## 算法思想

维护一个连续区间 `[l,r]`，根据当前窗口状态动态移动左右指针。

特点：

- 左右指针都只向右移动，不会回退。
- 每个元素最多进入、离开窗口一次。
- 时间复杂度为 **O(n)**。

---

## 适用条件

- 连续区间（子数组、子串）
- 窗口信息可以 O(1) 更新
- 满足单调性（扩大或缩小窗口后状态变化方向确定）

例如：

- 连续正整数和
- 最长/最短满足条件区间
- 最长不重复子串
- 最小覆盖区间

> **注意：**
>
> 若数组包含负数，普通滑动窗口通常不能直接使用。

---

## 窗口移动规则

设当前窗口和为 `sum`，目标值为 `target`。

```cpp
if(sum < target){
    r++;
    sum += 新元素;
}
else if(sum > target){
    sum -= 左端元素;
    l++;
}
else{
    更新答案;
    sum -= 左端元素;
    l++;
}
```

## 通用模板

```cpp
int l = ...;
int r = ...;

while(窗口仍然合法){

    if(满足条件){
        更新答案;
        缩小窗口;
    }
    else if(窗口太小){
        扩大窗口;
    }
    else{
        缩小窗口;
    }

}
```

## 经典题目
#### P1147 连续正整数和
```cpp
#include <bits/stdc++.h>
using namespace std;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int M;
    cin >> M;

    int l=1;
    int r=2;
    int sum=3;

    while(l<=M/2){
        if(sum==M){
            cout << l << " " << r << "\n";
            sum-=l;
            l++;
        }else if(sum < M){
            r++;
            sum+=r;
        }else if(sum > M){
            sum-=l;
            l++;
        }
    }
    

    return 0;
}
```

#### P1638 逛画展
滑动窗口思路：
```
// 1. 扩大窗口
...

// 2. 当窗口满足条件时
while(...){

    // 更新答案
    ...

    // 缩小窗口
    ...
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 1e6 + 5;
const int M = 2e3 + 5;
int a[N];
int cnt[M];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,m;
    cin >> n >> m;

    for(int i=1; i<=n; i++) cin >> a[i];

    int x = 1;
    int y = 1;
    int kind = 0;
    int l = 1;
    int r = 1;
    int ans = n+1;  //注意初始更新到无穷大后面才可以更新x,y

    while(r <= n){
        if(cnt[a[r]] == 0){
            kind++;
        }
        cnt[a[r]]++; 
        while(kind == m){
            if(r-l+1<ans || r-l+1==ans&&l<x){
                x = l;
                y = r;
                ans = r-l+1;
            }
           cnt[a[l]]--;
            if(cnt[a[l]]==0){
                kind --;
            }
            l++;
        }
        r++;
    }

    cout << x << " " << y << "\n";
    return 0;
}
```

####   P3143 [USACO16OPEN] Diamond Collector S
滑动窗口+最大后缀和
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 5e4 + 5;
int a[N];
int len[N];
int best[N];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,k;
    cin >> n >> k;

    for(int i=1; i<=n; i++)
        cin >> a[i];

    //首先排序，便于之后滑动窗口
    sort(a+1, a+n+1); //从第一个位置到第一个位置的下一个位置

    //滑动窗口并且计算更新len[i]:表示左端为i的窗口的大小
    int r=1;
    for(int l=1; l<=n; l++){
        while(r<=n && a[r] - a[l] <= k) r++;//存在越界
        len[l] = r-l;
    }

    //下面求best[i]数组，表示第i个位置到末尾最大的展示柜容纳数，需要逆序
    best[n] = len[n];
    for(int i=n-1; i>=1; i--){
        best[i] = max(len[i],best[i+1]);
    }

    //枚举更新最大值
    int ans = len[1]+best[1+len[1]];
    for(int i=2; i<=n-1; i++){
        int current = len[i] + best[i+len[i]];
        if(current > ans)
            ans = current;
    }

    cout << ans << "\n";
    return 0;
}
```

#### P1102[A-B 数对](https://www.luogu.com.cn/problem/P1102 "A-B 数对")
```cpp
#include <bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
long long a[N];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long c;
    cin >> n >> c;

    for(int i = 1; i <= n; i++)
        cin >> a[i];

    sort(a + 1, a + n + 1);

    int l = 1;
    int r = 1;
    long long ans = 0;

//a[r] - a[l] < C  → r++
//a[r] - a[l] > C  → l++
//a[r] - a[l] = C  → 统计两边重复次数并相乘
    
   while(l<=n && r<=n){  //在条件判断时候才左边或者是右边++
       if(l == r){
           r++;
           continue;
       }
       if(a[r]-a[l] > c){
           l++;
       }else if(a[r]- a[l] < c){
           r++;
       }else{
           long long l_cnt =0;
           long long r_cnt =0;

           int l_value=a[l];
           int r_value=a[r];

           while(l<=n && a[l]==l_value){
               l++;
               l_cnt++;
           }
           while(r<=n && a[r]==r_value){
               r++;
               r_cnt++;
           }

           ans+=l_cnt*r_cnt; //坑人：这里如果是int int那么相乘的时候就溢出来，所以l_cnt和r_cnt必须是long long
       }
   }
    cout << ans;
    return 0;
}
```
方法二：采用lower_bound upper_bound的二分[CSP算法/基础算法/二分]方法
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=2e5 + 5;
int a[N];

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,c;
    cin >> n >> c;

    for(int i=1; i<=n; i++) cin >> a[i];

    sort(a+1, a+n+1);

    long long ans = 0;
    
    for(int i=1; i<=n; i++){
        int target = a[i] + c;
        int left = lower_bound(a+1, a+n+1, target) - a;
        int right = upper_bound(a+1, a+n+1, target) - a;
        ans += right - left;
    }

    cout << ans;

    return 0;
}
```
#### P1714 [切蛋糕](https://www.luogu.com.cn/problem/P1714 "切蛋糕")
维护窗口的最小前缀和
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 5e5 + 5;
int sum[N];
const int NEG_INF = -505;

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,m;
    cin >> n >> m;

    int x;
    cin >> x;
    sum[0] = 0;
    sum[1] = x;
    for(int i=2; i<=n; i++){
        int x;
        cin >> x;
        sum[i] = sum[i-1] + x;
    }

    int ans = NEG_INF;
    deque<int> q;
    q.push_back(0);    //初始化
    for(int i=1; i<=n; i++){
        //目标找符合条件的最小sum[j]

        //首先删除不合理的角标
        while(!q.empty() && q.front() < i-m)
            q.pop_front();

        //计算最小sum[j]
        ans = max(ans, sum[i]-sum[q.front()]);

        //维护队列单调性并且当前入队
        while(!q.empty() && sum[i] <= sum[q.back()])
            q.pop_back();

        q.push_back(i);
    }
    cout << ans;
    return 0;
}
```
