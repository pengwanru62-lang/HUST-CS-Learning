#### P3367 [【模板】并查集](https://www.luogu.com.cn/problem/P3367 "【模板】并查集")（本质上是树形结构）（路径压缩 && 集合合并）
```cpp
#include<bits/stdc++.h>
using namespace std;

const int N = 2e5 + 5;
int parent[N];

int find_root(int x){
    if(parent[x] == x)
        return x;
    return parent[x] = find_root(parent[x]);    //依次将当前节点挂在根节点之下
}

void merge_set(int x, int y){
    int root_x = find_root(x);
    int root_y = find_root(y);

    if(root_x != root_y)
        parent[root_x] = root_y;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,m;
    cin >> n >> m;

    //初始化自己指向自己
    for(int i=1; i<=n; i++)
        parent[i] = i;

    //开始查询
    while(m--){
        int x, y, z;
        cin >> z;
        cin >> x >> y;
        if(z == 1){
            if(find_root(x) == find_root(y))
                continue;
            else
                merge_set(x, y);
        }else{
            if(find_root(x) == find_root(y))
                cout << 'Y' << endl;
            else
                cout << 'N' << endl;
        }
    }
    
    return 0;
}
```
#### P1955 [[NOI2015] 程序自动分析](https://www.luogu.com.cn/problem/P1955 "[NOI2015] 程序自动分析")（并查集 & 离散化）
```cpp
#include<bits/stdc++.h>
using namespace std;

struct Condition{
    int x;
    int y;
    int e;
};

const int N = 2e5 + 5;
int parent[N];

int find_root(int x){
    if(x == parent[x]){
        return x;
    }
    return parent[x] = find_root(parent[x]);
}

bool merge(int x, int y){
    int root_x = find_root(x);
    int root_y = find_root(y);

    if(root_x != root_y){
        parent[root_x] = root_y;
        return true;
    }
    return false;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while(t--){
        int n;
        cin >> n;

        vector<Condition> cdt;
        vector<int> num;

        for(int k=1; k<=n; k++){
            int i,j,e;
            cin >> i >> j >> e;
            cdt.push_back({i, j, e});
            
            num.push_back(i);
            num.push_back(j);
        }
        
        //排序
        sort(num.begin(), num.end());

        //去重
        num.erase(unique(num.begin(), num.end()), num.end());

        int cnt = num.size();

        //并查集初始化
        for(int i=0; i<cnt; i++){
            parent[i] = i;   //注意角标的匹配关系
        }

        //开始处理合并的部分
        for(auto condition : cdt){
            if(condition.e == 1){
                int x = lower_bound(num.begin(), num.end(), condition.x) - num.begin();
                int y = lower_bound(num.begin(), num.end(), condition.y) - num.begin();

                merge(x,y);
            }
        }

        //开始对比
        bool can = true;
        for(auto condition : cdt){
            if(condition.e == 0){
                int x = lower_bound(num.begin(), num.end(), condition.x) - num.begin();
                int y = lower_bound(num.begin(), num.end(), condition.y) - num.begin();

                int root_x = find_root(x);
                int root_y = find_root(y);

                if(root_x == root_y){
                    can = false;
                    break;
                }
            }
        }
        if(can)
            cout << "YES" << '\n';
        else
            cout << "NO" << '\n';
    }
    return 0;
}
```
